cmake_minimum_required(VERSION 3.28)
project(pcsx2_neural_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Test)
find_package(Vulkan REQUIRED)
enable_testing()
set(CONFIG_SOURCE ../../pcsx2-qt/Neural/neural_rendering_config.cpp)
add_executable(neural-config-tests config_tests.cpp ${CONFIG_SOURCE})
target_include_directories(neural-config-tests PRIVATE ../../pcsx2-qt/Neural)
target_link_libraries(neural-config-tests PRIVATE Qt6::Core)
add_test(NAME neural-config COMMAND neural-config-tests)
add_executable(neural-ui-tests ui_tests.cpp ${CONFIG_SOURCE} ../../pcsx2-qt/Settings/NeuralRenderingSettingsWidget.cpp)
target_include_directories(neural-ui-tests PRIVATE ../../pcsx2-qt ../../pcsx2-qt/Neural)
target_link_libraries(neural-ui-tests PRIVATE Qt6::Widgets Qt6::Test)
add_test(NAME neural-ui COMMAND neural-ui-tests)
set_tests_properties(neural-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(neural-vulkan-cycle vulkan_cycle.cpp ${CONFIG_SOURCE})
target_include_directories(neural-vulkan-cycle PRIVATE ../../pcsx2-qt/Neural)
target_link_libraries(neural-vulkan-cycle PRIVATE Qt6::Core Vulkan::Vulkan user32)
foreach(target neural-config-tests neural-ui-tests neural-vulkan-cycle)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8)
    endif()
endforeach()
# The GPU cycle test is opt-in: run only in a disposable directory containing
# the locally downloaded runtime. No proprietary runtime is part of this tree.
